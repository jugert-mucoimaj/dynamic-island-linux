cmake_minimum_required(VERSION 3.16)
project(Island VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets DBus)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets DBus)
if(QT_VERSION_MAJOR EQUAL 6)
    find_package(Qt6 REQUIRED COMPONENTS OpenGLWidgets)
    set(ISLAND_GL Qt6::OpenGLWidgets)
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0)
pkg_check_modules(XSHAPE REQUIRED IMPORTED_TARGET x11 xext)
set(ISLAND_SOURCES src/iconbutton.h src/island.h src/activities.h src/notifications.h src/notifications.cpp src/playbackindicator.h src/playbackindicator.cpp src/audiomonitor.cpp src/audiomonitor.h src/inputregion.cpp src/inputregion.h)
add_executable(island src/main.cpp ${ISLAND_SOURCES})
target_link_libraries(island PRIVATE Qt${QT_VERSION_MAJOR}::Widgets Qt${QT_VERSION_MAJOR}::DBus PkgConfig::GIO PkgConfig::XSHAPE ${ISLAND_GL})
target_compile_options(island PRIVATE -Wall -Wextra -Wpedantic)
add_executable(island-run src/runner.cpp)
target_link_libraries(island-run PRIVATE Qt${QT_VERSION_MAJOR}::Core Qt${QT_VERSION_MAJOR}::DBus)
target_compile_options(island-run PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS island island-run RUNTIME DESTINATION bin)

include(CTest)
if(BUILD_TESTING)
    find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Test)
    add_executable(island_tests tests/island_test.cpp ${ISLAND_SOURCES})
    target_include_directories(island_tests PRIVATE src)
    target_link_libraries(island_tests PRIVATE Qt${QT_VERSION_MAJOR}::Widgets Qt${QT_VERSION_MAJOR}::DBus Qt${QT_VERSION_MAJOR}::Test PkgConfig::GIO PkgConfig::XSHAPE ${ISLAND_GL})
    target_compile_definitions(island_tests PRIVATE ISLAND_RUN_PATH="$<TARGET_FILE:island-run>")
    add_dependencies(island_tests island-run)
    find_program(DBUS_RUN_SESSION dbus-run-session REQUIRED)
    add_test(NAME island_tests COMMAND ${DBUS_RUN_SESSION} -- $<TARGET_FILE:island_tests>)
    set_tests_properties(island_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;ISLAND_DISABLE_AUDIO=1")
endif()
